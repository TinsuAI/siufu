### Google Cloud Document AI API Details

[Source: architecture/external-apis.md#google-cloud-document-ai-api]

**API Endpoint:**
```
POST /v1/projects/{project}/locations/{location}/processors/{processor}:process
```

**Request Structure:**
```python
from google.cloud import documentai

# Initialize client
client = documentai.DocumentProcessorServiceClient()

# Processor path
name = f"projects/{project_id}/locations/{location}/processors/{processor_id}"

# Read document
with open(file_path, 'rb') as f:
    document_content = f.read()

# Create request
request = documentai.ProcessRequest(
    name=name,
    raw_document=documentai.RawDocument(
        content=document_content,
        mime_type="application/pdf"  # or "image/jpeg"
    )
)

# Process document
response = client.process_document(request=request)

# Access results
text = response.document.text
entities = response.document.entities  # key-value pairs
tables = response.document.pages[0].tables  # tables per page
```

**Response Structure:**
```python
response.document:
  .text: str  # Full OCR text
  .entities: List[Entity]  # Key-value pairs
    .type: str  # Entity type (e.g., "invoice_date")
    .mention_text: str  # Extracted value
    .confidence: float  # 0.0-1.0
  .pages: List[Page]
    .tables: List[Table]
      .header_rows: List[Row]
      .body_rows: List[Row]
        .cells: List[Cell]
          .text: str
  .confidence_scores: float  # Overall confidence
```

**Confidence Scores:**
- Overall document confidence: `response.document.confidence_scores`
- Per-entity confidence: `entity.confidence`
- Per-table confidence: Average of cell confidences

from pydantic import BaseModel, Field
from typing import List, Dict

class KeyValuePair(BaseModel):
    key: str
    value: str
    confidence: float = Field(ge=0.0, le=1.0)

class TableCell(BaseModel):
    text: str
    confidence: float

class TableRow(BaseModel):
    cells: List[TableCell]

class Table(BaseModel):
    headers: List[str]
    rows: List[List[str]]  # Simplified: list of row values
    confidence: float

class OCRResult(BaseModel):
    text: str  # Full extracted text
    key_value_pairs: List[KeyValuePair]
    tables: List[Table]
    confidence_scores: Dict[str, float]  # field_name -> confidence
    page_count: int
    file_name: str
    processing_time_ms: int  # For cost tracking

    model_config = {"json_schema_extra": {
        "example": {
            "text": "ARRIVAL NOTICE\\nVessel: MSC GEMMA...",
            "key_value_pairs": [
                {"key": "vessel_name", "value": "MSC GEMMA", "confidence": 0.95}
            ],
            "tables": [
                {
                    "headers": ["Container", "Size", "Weight"],
                    "rows": [["MSCU1234567", "40'", "20,000 kg"]],
                    "confidence": 0.92
                }
            ],
            "confidence_scores": {"vessel_name": 0.95, "container_number": 0.92},
            "page_count": 1,
            "file_name": "AN.pdf",

### Redis Caching Implementation

[Source: Acceptance Criteria 5, architecture/tech-stack.md]

**Cache Key Strategy:**
```python
import hashlib

def get_ocr_cache_key(file_path: str) -> str:
    """Generate cache key using MD5 hash of file contents + file path"""
    with open(file_path, 'rb') as f:
        file_hash = hashlib.md5(f.read()).hexdigest()
    return f"ocr:result:{file_hash}"
```

**Cache Storage:**
```python
import redis
import json

# Serialize OCRResult to JSON
cache_value = ocr_result.model_dump_json()

# Store in Redis with 24-hour TTL (86400 seconds)
redis_client.setex(cache_key, 86400, cache_value)
```

**Cache Retrieval:**
```python
# Get cached result
cached_value = redis_client.get(cache_key)

if cached_value:
    # Deserialize JSON to OCRResult
    ocr_result = OCRResult.model_validate_json(cached_value)
    return ocr_result
```

- Track cache hit rate: `cache_hits / (cache_hits + cache_misses)`

**Retry Logic with Tenacity:**
```python
from tenacity import retry, stop_after_attempt, wait_exponential
import sentry_sdk

@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=1, max=4),
    reraise=True
)
def _call_document_ai_api(request):
    """Call Document AI API with retry logic"""
    try:
        return client.process_document(request=request)
    except google.api_core.exceptions.GoogleAPIError as e:
        sentry_sdk.capture_exception(e)
        raise
```

import os
import json
from google.oauth2 import service_account

def load_gcp_credentials():
    """Load GCP service account credentials from Docker secret"""
    secret_path = "/run/secrets/gcp-sa-key"

    if not os.path.exists(secret_path):
        raise FileNotFoundError(
            f"GCP service account key not found at {secret_path}. "
            "Please mount secret in docker-compose.yml. "
            "See docs/setup-google-cloud.md for instructions."

        credentials_info
    )

    return credentials

import sentry_sdk
import time

def process_document_ocr(file_path: str) -> OCRResult:
    start_time = time.time()

    # Check cache first
    cached_result = get_cached_ocr_result(file_path)
    if cached_result:
        sentry_sdk.add_breadcrumb(
            category="ocr",
            message=f"Cache hit for {file_path}",
            level="info"
        )
        return cached_result

    # Call API
    result = _call_document_ai_api(file_path)

    # Log usage

    sentry_sdk.set_measurement("document_ai_pages", result.page_count)
    sentry_sdk.set_measurement("document_ai_processing_ms", processing_time)
    sentry_sdk.set_tag("document_ai_file", os.path.basename(file_path))

    # Cache result
    cache_ocr_result(file_path, result)

    return result
